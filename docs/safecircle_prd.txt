4. Final 60-second pitch script that wins hackathons

“Digital safety for young users is usually handled through surveillance: centralized tracking,
invasive monitoring, and opaque data practices. We think that’s the wrong model.
We built SafeCircle, a privacy-first safety layer for youth digital platforms. Our system
detects harmful patterns like bullying, coercion, or grooming locally on the user’s device, so
raw conversations never need to leave the device.

model updates are shared, not private messages. We also minimize metadata exposure and

contents. This means parental guidance without invasive surveillance.
ExpressVPN protects privacy at the network layer. SafeCircle extends that same privacy-first
philosophy to youth safety at the application layer: safety without surveillance.”
That is strong, clean, and directly aligned.

from collections import Counter
from .models import PrivacySignal
def guardian_summary(signals: list[PrivacySignal]) -> dict:
    if not signals:
        return {
            "risk_level": "low",
            "top_categories": [],
            "event_count": 0,
        }
    severities = [s.severity for s in signals]
    categories = [s.category for s in signals]
    if "high" in severities:
        risk_level = "high"
    elif severities.count("medium") >= 2:
        risk_level = "medium"
    else:
        risk_level = "low"
    top_categories = Counter(categories).most_common(3)
    return {
        "risk_level": risk_level,
        "top_categories": top_categories,
        "event_count": len(signals),
    }

class SafeCircleClient:
    def __init__(self, api_url: str = "http://127.0.0.1:8000/predict"):
        self.api_url = api_url
    def _infer_reasons(self, text: str, label_name: str) -> list[str]:
        lower = text.lower()
        reasons = []
        if "don't tell" in lower or "dont tell" in lower:
            reasons.append("secrecy_request")
        if "meet me" in lower:


## Page 27
            reasons.append("isolated_meetup")
        if "worthless" in lower or "stupid" in lower or "nobody likes you" in lower:
            reasons.append("harassment_language")
        if "alone" in lower:
            reasons.append("isolation_language")
        if not reasons and label_name == "unsafe":
            reasons.append("unsafe_pattern")
        if not reasons and label_name == "safe":
            reasons.append("safe")

    def analyze_text(self, text: str) -> AnalysisResult:
        resp = requests.post(self.api_url, json={"text": text}, timeout=10)
